function doPost(e) {
  const output = ContentService.createTextOutput().setMimeType(ContentService.MimeType.JSON);
  try {
    const payload = JSON.parse(e.postData.contents);
    const action = String(payload.action || '').trim();
    
    // Log de acciones (excepto las más frecuentes para no saturar)
    const silentActions = ['listarPedidos', 'getCajaData', 'obtenerHorariosSemanales', 'obtenerLlavesExistentes'];
    if (!silentActions.includes(action)) {
      logToSheet("Action: " + action + " | ID: " + (payload.nro || payload.llave || 'n/a'));
    }
    
    let result = {};
    switch(action) {
      case 'login':               result = login(payload.user, payload.pass); break;
      case 'listarPedidos':       result = listarPedidos(); break;
      case 'crearPedido':         result = crearPedido(payload); break;
      case 'validarPedido':       result = validarPedido(payload); break;
      case 'rechazarPedido':      result = rechazarPedido(payload); break;
      case 'crearPedidosMasivos': result = crearPedidosMasivos(payload); break;
      case 'actualizarHorasTada': result = actualizarHorasTada(payload.orders); break;
      case 'marcarSLAFuera':      result = marcarSLAFuera(payload.nro, payload.fuera_sla); break;
      case 'guardarFotoOcr':      result = guardarFotoOcrSilent(payload.llave, payload.fotoBase64, payload.lat, payload.lng, payload.accuracy, payload.phoneTimestamp); break;
      case 'processVoucherOCR':   result = processVoucherOCR(payload); break;
      case 'setGeminiKey':        result = setGeminiKey(payload); break;
      case 'loginMotorizado':     result = loginMotorizado(payload.user, payload.pass); break;
      case 'marcarPorValidar':    result = marcarPorValidar(payload.nro, payload.fechaEntrega, payload.horaEntrega, payload.lat, payload.lng, payload.accuracy, payload.phoneTimestamp, payload.isAutoValidated); break;
      case 'asignarMotorizado':   result = asignarMotorizado(payload); break;
      case 'marcarEnCamino':      result = marcarEnCamino(payload.nro); break;
      case 'crearViajeAutomatico': result = crearViajeAutomatico(payload); break;
      case 'guardarOrdenRutaMotorizado': result = guardarOrdenRutaMotorizado(payload); break;
      case 'asignarViajePedido':  result = asignarViajePedido(payload); break;
      case 'corregirEntrega':     result = corregirEntrega(payload); break;
      case 'getPhotoBase64':      result = getPhotoBase64(payload.nro); break;
      case 'crearMotorizado':     result = crearMotorizado(payload); break;
      case 'guardarHorarioSemananal': result = guardarHorarioSemananal(payload); break;
      case 'obtenerHorariosSemanales': result = obtenerHorariosSemanales(payload.semanaId); break;
      case 'registrarMovimientoCaja': result = registrarMovimientoCaja(payload); break;
      case 'getCajaData':             result = getCajaData(payload); break;
      case 'guardarVueltoInline':     result = guardarVueltoInline(payload); break;
      case 'obtenerNombresMotorizados': result = obtenerNombresMotorizados(); break;
      case 'obtenerLlavesExistentes':   result = obtenerLlavesExistentes(); break;
      case 'extraerListaPOS':           result = extraerListaPOS(payload); break;
      case 'guardarAuditoriaPOS':       result = guardarAuditoriaPOS(payload); break;
      case 'obtenerAuditoriasPOS':      result = obtenerAuditoriasPOS(payload.filtro); break;
      default:                    result = { success: false, message: 'Acción no encontrada' };
    }
    return output.setContent(JSON.stringify(result));
  } catch (err) {
    return output.setContent(JSON.stringify({ success: false, message: err.toString() }));
  }
}

function getSheet() {

function login(user, pass) {
  const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName('Usuarios');
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) == user && String(data[i][1]) == pass) {
      return { success: true, user: { usuario: data[i][0], nombre: data[i][2], rol: data[i][3] || 'Lector' } };
    }
  }
  return { success: false, message: "Credenciales incorrectas" };
}

function listarPedidos() {
  const sheet = getSheet();
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return { success: true, data: [] };
  const data = sheet.getRange(2, 1, lastRow - 1, 32).getValues();
  const displayData = sheet.getRange(2, 1, lastRow - 1, 32).getDisplayValues();

  const pedidos = data.map((row, index) => ({
    nro: row[0], fecha: row[1], llave: row[2], monto: row[3], foto: row[4],
    monto_foto: row[5], estado: row[6], validado_por: row[7], fecha_validacion: row[8],
    envio: row[9], pago: row[10], tipo_pago: row[11], vuelto: row[12], motivo_cancelacion: row[13],
    sla_fuera: row[14], fecha_entrega: displayData[index][15], hora_entrega: displayData[index][16],
    tiempo_transcurrido: displayData[index][17], orden_ruta: row[18], viaje_id: row[19],
    idTransaccion: row[26], hallazgoRobot: row[31]
  })).filter(p => p.nro);

  return { success: true, data: pedidos.sort((a, b) => b.nro - a.nro) };
}

// --- LÓGICA DE VALIDACIÓN Y AUTOMATIZACIÓN ---

function validarPedido(data) {
  const sheet = getSheet();
  const rows = sheet.getDataRange().getValues();
  let rowIndex = -1;
  for (let i = 1; i < rows.length; i++) {
    if (rows[i][0] == data.nro) { rowIndex = i + 1; break; }
  }
  if (rowIndex === -1) return { success: false, message: "Pedido no encontrado" };

  // 1. Actualizar Pedido
  sheet.getRange(rowIndex, 7).setValue("Validado");
  sheet.getRange(rowIndex, 8).setValue(data.usuario);
  sheet.getRange(rowIndex, 9).setValue(new Date());
  sheet.getRange(rowIndex, 12).setValue(data.tipo || "");
  sheet.getRange(rowIndex, 13).setValue(data.vuelto || "");
  if (data.fechaEntrega) sheet.getRange(rowIndex, 16).setValue(data.fechaEntrega);
  if (data.horaEntrega) sheet.getRange(rowIndex, 17).setValue(data.horaEntrega);

  // 2. AUTOMATIZACIÓN DE CAJA (Solo si es Efectivo/Contado)
  const pagoReal = (data.tipo || "").toString().toUpperCase();
  if (pagoReal.includes("EFECTIVO") || pagoReal.includes("CONTADO")) {
    try {
      const montoVuelto = parseFloat(data.vuelto) || 0;
      const montoPedido = parseFloat(rows[rowIndex-1][3]) || 0;
      const repartidor = (data.envio || rows[rowIndex-1][9] || "").toString().trim();
      
      // Auto-Egreso del Vuelto
      if (montoVuelto > 0) {
        registrarMovimientoCaja({
          tipo: 'EGRESO', metodo: 'FISICO',
          concepto: "Vuelto Automático Pedido #" + data.nro + " - " + repartidor,
          monto: montoVuelto, pedidoNro: data.nro, usuario: data.usuario
        });
      }
      // Auto-Ingreso del Cobro (Monto + Vuelto)
      registrarMovimientoCaja({
        tipo: 'INGRESO', metodo: 'FISICO',
        concepto: "Cobro Automático Pedido #" + data.nro + " - " + repartidor,
        monto: montoPedido + montoVuelto, pedidoNro: data.nro, usuario: data.usuario
      });
    } catch(e) { logToSheet("Error Caja Auto: " + e.toString()); }
  }

  return { success: true };
}

function marcarPorValidar(nro, fEnt, hEnt, lat, lng, acc, ts, isAuto) {
  const sheet = getSheet();
  const rows = sheet.getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]).trim() == String(nro).trim()) {
      sheet.getRange(i + 1, 7).setValue('Por Validar');
      if (fEnt) sheet.getRange(i + 1, 16).setValue(fEnt);
      if (hEnt) sheet.getRange(i + 1, 17).setValue(hEnt);
      if (lat) sheet.getRange(i + 1, 21).setValue(lat);
      if (lng) sheet.getRange(i + 1, 22).setValue(lng);
      return { success: true };
    }
  }
  return { success: false };
}

// --- GESTIÓN DE CAJA CHICA ---

function ensureCajaSheet() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  let sheet = ss.getSheetByName('Caja_Movimientos') || ss.insertSheet('Caja_Movimientos');
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['ID', 'Fecha', 'Hora', 'Tipo', 'Concepto', 'Monto', 'Pedido_Nro', 'Repartidor', 'Usuario', 'Metodo', 'Saldo_Fisico', 'Saldo_Digital']);
  }
  return sheet;
}

function registrarMovimientoCaja(data) {
  try {
    const sheet = ensureCajaSheet();
    const lastRow = sheet.getLastRow();
    let sF = 0, sD = 0;
    if (lastRow >= 2) {
      const last = sheet.getRange(lastRow, 11, 1, 2).getValues()[0];
      sF = parseFloat(last[0]) || 0;
      sD = parseFloat(last[1]) || 0;
    }
    const monto = parseFloat(data.monto) || 0;
    if (data.metodo === "DIGITAL") sD += (data.tipo === "INGRESO" ? monto : -monto);
    else sF += (data.tipo === "INGRESO" ? monto : -monto);
    
    const now = new Date();
    sheet.appendRow([Date.now(), Utilities.formatDate(now, "GMT-5", "dd/MM/yyyy"), Utilities.formatDate(now, "GMT-5", "HH:mm:ss"), data.tipo, data.concepto, monto, data.pedidoNro, data.repartidor, data.usuario, data.metodo || "FISICO", sF, sD]);
    return { success: true };
  } catch(e) { return { success: false, message: e.toString() }; }
}

function getCajaData(params) {
  try {
    const sheet = ensureCajaSheet();
    const data = sheet.getDataRange().getValues();
    const disp = sheet.getDataRange().getDisplayValues();
    const movs = data.slice(1).map((r, i) => ({
      id: r[0], fecha: disp[i+1][1], hora: disp[i+1][2], tipo: r[3], concepto: r[4], monto: r[5], pedidoNro: r[6], saldoFisico: r[10], saldoDigital: r[11]
    })).reverse();
    return { success: true, movimientos: movs.slice(0, 50), saldoFisico: movs[0]?.saldoFisico || 0, saldoDigital: movs[0]?.saldoDigital || 0 };
  } catch(e) { return { success: false }; }
}

function obtenerHorariosSemanales(semanaId) {

function guardarHorarioSemananal(p) {
  try {
    const sheet = getHorariosSheet();
    const data = sheet.getDataRange().getValues();
    for (let i = data.length - 1; i >= 1; i--) if (data[i][0] === p.semanaId) sheet.deleteRow(i + 1);
    const rows = p.horarios.map(h => [p.semanaId, h.nro, h.status, h.driver, h.lunes, h.martes, h.miercoles, h.jueves, h.viernes, h.sabado, h.domingo]);
    if (rows.length) sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, 11).setValues(rows);
    return { success: true };
  } catch(e) { return { success: false }; }
}

// --- GEMINI OCR Y DRIVE ---

function processVoucherOCR(data) {
  const apiKey = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY');
  if (!apiKey) return { success: false, message: "Falta API Key" };
  const prompt = 'Extrae de este voucher: {"fecha":"DD/MM/YY","hora":"HH:MM","total":0.00,"tipoPago":"QR|TARJETA","idOperacion":""}. Devuelve SOLO JSON.';
  const url = 'https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=' + apiKey;
  const payload = { contents: [{ parts: [{ text: prompt }, { inline_data: { mime_type: data.mimeType, data: data.imageBase64 } }]}] };
  try {
    const res = UrlFetchApp.fetch(url, { method: 'post', contentType: 'application/json', payload: JSON.stringify(payload) });
    const text = JSON.parse(res.getContentText()).candidates[0].content.parts[0].text.replace(/```json|```/g, '').trim();
    return { success: true, data: JSON.parse(text) };
  } catch(e) { return { success: false, message: e.toString() }; }
}

function getPhotoBase64(nro) {
  try {
    const sheet = getSheet();
    const rows = sheet.getDataRange().getValues();
    let url = "";
    for (let i = 1; i < rows.length; i++) if (rows[i][0] == nro) { url = rows[i][4]; break; }
    if (!url) return { success: false };
    const id = url.match(/\/d\/([a-zA-Z0-9_-]+)/)[1];
    const blob = DriveApp.getFileById(id).getBlob();
    return { success: true, base64: Utilities.base64Encode(blob.getBytes()), mimeType: blob.getContentType() };
  } catch(e) { return { success: false }; }
}
